// 初始化主题 const saved = localStorage.getItem('ice-theme') || 'auto'; let theme = saved === 'auto' ? (window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light') : saved; document.documentElement.setAttribute('data-theme', theme); // ⚠️ 替换成你真实的 Worker 链接 const API_BASE = 'https://ice-apis.ryokn0ce.workers.dev'; const urlParams = new URLSearchParams(window.location.search); const targetPage = urlParams.get('from') || 'main.html'; let isForceLogin = false; // 是否强制执行踢下线 document.getElementById('loginBtn').addEventListener('click', async function() { const user = document.getElementById('username').value.trim(); const pass = document.getElementById('password').value.trim(); const errorMsg = document.getElementById('errorMsg'); if (!user || !pass) { errorMsg.textContent = "用户名和密码不能为空"; errorMsg.style.display = 'block'; return; } const btn = this; btn.disabled = true; btn.textContent = "验证中..."; errorMsg.style.display = 'none'; try { // 如果是强制登录，走 force-login 接口 const endpoint = isForceLogin ? '/api/force-login' : '/api/login'; const res = await fetch(API_BASE + endpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: user, password: pass }) }); const data = await res.json(); if (res.ok) { // 验证成功，保存 Token localStorage.setItem('ice_user', user); localStorage.setItem('ice_token', data.token); // 通知导航栏 try { window.parent.document.getElementById('navFrame').contentWindow.postMessage({ type: 'updateAuth' }, '*'); } catch (e) {} // 跳转 try { window.parent.postMessage({ type: 'navigate', url: targetPage }, '*'); } catch (e) { window.location.href = '../' + targetPage; } } else if (res.status === 409 && !isForceLogin) { // 触发互踢检测 errorMsg.style.color = '#ff9f0a'; // 警告色 errorMsg.innerHTML = data.message + '
点此强制下线该设备并重新登录'; errorMsg.style.display = 'block'; document.getElementById('forceBtn').onclick = function() { isForceLogin = true; btn.click(); // 重新触发登录 }; } else { errorMsg.style.color = '#ff453a'; errorMsg.textContent = data.error || "认证失败"; errorMsg.style.display = 'block'; } } catch (err) { errorMsg.style.color = '#ff453a'; errorMsg.textContent = "网络连接失败 (Worker 离线)"; errorMsg.style.display = 'block'; } finally { btn.disabled = false; btn.textContent = "验证并进入"; } }); document.getElementById('cancelBtn').addEventListener('click', function(e) { e.preventDefault(); try { window.parent.postMessage({ type: 'navigate', url: targetPage }, '*'); } catch (err) { window.location.href = '../' + targetPage; } });
